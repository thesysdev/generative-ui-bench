```openui
root = Card(
    [
        header,
        stepsSection,
        caloriesSection,
        workoutsSection,
        logWorkoutButton,
    ],
    "card",
    "row",
    "m",
    "center",
    "between",
    wrap = true
)

header = CardHeader("Daily Fitness Summary", "Today's Progress")

stepsSection = Card(
    [
        stepsHeader,
        stepsProgress,
    ],
    "sunk",
    direction = "column",
    gap = "s"
)
stepsHeader = Heading("Steps", "h3")
stepsProgress = Card(
    [
        stepsValue,
        stepsGoal,
    ],
    "clear",
    direction = "row",
    justify = "between",
    align = "center"
)
stepsValue = TextContent("12,400 steps", "large")
stepsGoal = TextContent("Target: 10,000 steps", "small")

caloriesSection = Card(
    [
        caloriesHeader,
        caloriesValue,
    ],
    "sunk",
    direction = "column",
    gap = "s"
)
caloriesHeader = Heading("Calories Burned", "h3")
caloriesValue = TextContent("1,845 kcal", "large")

workoutsSection = Card(
    [
        workoutsHeader,
        workoutsChart,
    ],
    "sunk",
    direction = "column",
    gap = "m"
)
workoutsHeader = Heading("Past Week Workouts", "h3")
workoutsChart = BarChart(
    ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"],
    [
        Series("Workout Duration (min)", [45, 60, 30, 55, 50, 70, 40]),
        Series("Intensity (Low/Medium/High)", ["Medium", "High", "Low", "Medium", "High", "High", "Low"])
    ],
    variant = "grouped",
    xLabel = "Day",
    yLabel = "Workout"
)

logWorkoutButton = Button("Log New Workout", "primary")

root = Card(
    [
        header,
        stepsSection,
        caloriesSection,
        workoutsSection,
        logWorkoutButton,
    ],
    "card",
    "row",
    "m",
    "center",
    "between",
    wrap = true
)
```
