```openui-lang
root = Card([
    headline,
    emailForm,
    subscribeButton
])

headline = TextContent(
    "Subscribe to our weekly product newsletter and get the latest updates, tips, and exclusive sneak peeks delivered straight to your inbox every Friday.",
    "large"
)

emailForm = Form(
    "newsletter-signup",
    Buttons([]),  // placeholder, no explicit action button needed here
    [
        emailField
    ]
)

emailField = FormControl(
    "Email address",
    emailInput,
    "Enter your email to receive weekly updates"
)

emailInput = Input(
    "email",
    placeholder: "you@example.com",
    type: "email"
)

subscribeButton = Button(
    "Subscribe weekly",
    action: ["newsletter", "subscribe", emailInput.value],
    variant: "primary"
)
```
