root = Card([
    jobHeader,
    roleDescription,
    qualifications,
    salaryRange,
    applyForm,
    shareButton
], "sunk")

jobHeader = CardHeader("Product Designer", "UX Team")
roleDescription = TextContent(
    "We're looking for a talented Product Designer to join our UX Team. You'll shape the user experience of our flagship product, collaborating closely with engineers and product managers to deliver intuitive, elegant solutions."
)
qualifications = Card([mainQualifications, additionalQualifications])
mainQualifications = TextContent(
    "Key qualifications include:\n- 3+ years designing consumer-facing products\n- Proficiency in Figma and Adobe Creative Suite\n- Strong understanding of UX research methodologies"
)
additionalQualifications = TextContent(
    "- Experience with interactive prototyping\n- Ability to translate user feedback into design iterations\n- Excellent communication and collaboration skills"
)
salaryRange = Tag("USD 90,000 - 110,000", null, null, "success")
applyForm = Form("apply", applyButtons, [
    nameInput,
    emailInput,
    resumeLink,
    submitButton
])
nameInput = FormControl("Full Name", nameInputField)
nameInputField = Input("name", placeholder="Jane Smith")
emailInput = FormControl("Work Email", emailInputField)
emailInputField = Input("email", placeholder="jane@company.com", type="email")
resumeLink = FormControl("Resume Link", resumeLinkField)
resumeLinkField = Input("resume_url", placeholder="https://example.com/resume.pdf", type="url")
submitButton = Button("Apply Now", null, "primary")
applyButtons = Buttons([submitButton])

shareButton = Button("Share this posting", null, "secondary")
