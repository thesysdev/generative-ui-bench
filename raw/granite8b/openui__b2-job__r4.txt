```openui
root = Card([
    header,
    description,
    qualifications,
    salary,
    applyForm,
    shareButton
], "sunk")

header = CardHeader("Senior UI/UX Designer", "Design Team")
description = TextContent(
    "We are looking for a Senior UI/UX Designer to join our Design Team and help shape the user experience of our flagship product. The ideal candidate will have a strong portfolio showcasing interaction design, usability testing, and a collaborative mindset."
)
qualifications = Card([req1, req2, req3, req4], "clear")
req1 = TextContent("5+ years of UI/UX design experience with a focus on web and mobile interfaces.")
req2 = TextContent("Proficiency in design tools such as Figma, Sketch, or Adobe XD.")
req3 = TextContent("Strong understanding of user research methodologies and usability testing.")
req4 = TextContent("Ability to translate user stories into high-fidelity prototypes and handoff assets to developers.")
salary = TextContent("Salary Range: $110,000 - $150,000 per year")
applyForm = Form("design-application", applyButtons, [
    nameField,
    emailField,
    resumeField
])
nameField = FormControl("Full Name", nameInput)
nameInput = Input("name", placeholder: "Jane Smith")
emailField = FormControl("Work Email", emailInput, hint: "We will only use this to contact you about the role.")
emailInput = Input("email", placeholder: "jane@company.com", type: "email")
resumeField = FormControl("Resume Link", resumeInput)
resumeInput = Input("resume", placeholder: "https://example.com/resume.pdf", type: "url")
applyButtons = Buttons([
    applyBtn,
    cancelBtn
])
applyBtn = Button("Apply Now", null, "primary")
cancelBtn = Button("Cancel", null, "tertiary")
shareButton = Button("Share Job Posting", null, "secondary")
```
