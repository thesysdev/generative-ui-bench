dateRangePicker = DatePicker("date_range", mode="range", defaultMonth="2025-08", numberOfMonths=1, captionLayout="label")

root = Card(
    [
        CardHeader("Weekly Marketing Dashboard", "Last 30 Days"),
        dateRangePicker,
        visitorsChart,
        conversionRate,
        trafficSources,
        campaignsTable,
        bestCampaignHighlight,
        emailGrowth,
        exportBtn
    ],
    variant="card",
    direction="row",
    gap="m",
    align="center",
    justify="between",
    wrap=false
)

        Col("Share (%)", [25, 40, 15, 10, 10], type="number")

)

    columns=[
        Col("Campaign", ["Campaign A", "Campaign B", "Campaign C", "Campaign D", "Campaign E"], type="string"),
        Col("Spend ($)", [5000, 7500, 3000, 12000, 8000], type="number"),
