root = Card(

        responsibilities,

    "card",
    "column",

    "between",
    false,
)

    resumeLink,
    "Share a link to your resume or portfolio.",

    "resume_link",
    placeholder="https://example.com/resume.pdf",
    type="url",

        cancelBtn,
