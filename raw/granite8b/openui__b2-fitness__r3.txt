root = Card([
    dailySummaryHeader,
    stepProgress,
    caloriesBurned,
    weeklyWorkoutSummary,
    logNewWorkoutForm,
    footerButtons
], "card", "column", "none", "center", "between")

dailySummaryHeader = CardHeader("Daily Fitness Summary", "Today's Progress")

stepProgress = Card([
    stepCountLabel,
    stepProgressIndicator,
    stepGoalLabel
])
stepCountLabel = TextContent("Steps taken: 7,842", "large")
stepProgressIndicator = Progress(value=78.6, label="78% of goal")
stepGoalLabel = TextContent("Daily goal: 10,000 steps", "small")

caloriesBurned = Card([
    caloriesTitle,
    caloriesValue,
    caloriesExplanation
])
caloriesTitle = Heading("Calories burned", "h3")
caloriesValue = TextContent("Calories burned: 3,450 kcal", "large")
caloriesExplanation = TextContent("Includes activity and basal metabolism", "small")

weeklyWorkoutSummary = Card([
    weeklyWorkoutsTitle,
    workoutTable
])
weeklyWorkoutsTitle = Heading("Workouts this week", "h3")
workoutTable = Table([
    dayCol,
    durationCol,
    intensityCol,
    notesCol
])
dayCol = Col("Day", ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"])
durationCol = Col("Duration (min)", ["45", "30", "60", "40", "50", "35", "55"])
intensityCol = Col("Intensity", ["High", "Medium", "High", "Medium", "High", "Low", "Medium"])
notesCol = Col("Notes", ["Chest day", "Cardio", "Legs", "Core", "Rest", "Stretching", "Yoga"])

logNewWorkoutForm = Form("log-workout", formButtons, [
    workoutTypeField,
    durationField,
    intensityField,
    notesField
])
workoutTypeField = FormControl("Workout type", workoutTypeSelect)
workoutTypeSelect = Select("workout_type", [typeCardio, typeStrength, typeYoga], "Choose type")
typeCardio = SelectItem("Cardio", "Cardio")
typeStrength = SelectItem("Strength", "Strength training")
typeYoga = SelectItem("Yoga", "Yoga/Pilates")

durationField = FormControl("Duration (minutes)", durationInput)
durationInput = Input("duration", "45", "number")

intensityField = FormControl("Intensity", intensitySelect)
intensitySelect = Select("intensity", [intHigh, intMedium, intLow], "Choose intensity")
intHigh = SelectItem("High", "High")
intMedium = SelectItem("Medium", "Medium")
intLow = SelectItem("Low", "Low")

notesField = FormControl("Notes", notesTextArea)
notesTextArea = TextArea("notes", "Feel great!", 4)

formButtons = Buttons([
    submitBtn,
    cancelBtn
])
submitBtn = Button("Log workout", null, "primary")
cancelBtn = Button("Cancel", null, "tertiary")

footerButtons = Buttons([
    exportBtn,
    shareBtn
], "column")
exportBtn = Button("Export summary", null, "secondary")
shareBtn = Button("Share report", null, "tertiary")
