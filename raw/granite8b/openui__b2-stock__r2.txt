```openui-lang
root = Card([
    header,
    itemsTable,
    reorderForm,
    searchInput
], "sunk")

header = CardHeader("Low Inventory Warning", "Items below reorder threshold")

itemsTable = Table([
    colName,
    colUnits,
    colThreshold,
    colAction
])

colName = Col("Item", ["Widget A", "Gadget B", "Tool C"], "string")
colUnits = Col("Remaining Units", [12, 5, 8], "number")
colThreshold = Col("Reorder Threshold", [10, 10, 10], "number")
colAction = Col("Reorder", ["Reorder", "Reorder", "Reorder"], "action")

reorderForm = Form("reorder-form", reorderButtons, [
    formItemName,
    formItemUnits,
    formItemThreshold,
    formItemSearch
])

formItemName = FormControl("Item Name", nameInput)
nameInput = Input("name", placeholder="Widget A")

formItemUnits = FormControl("Current Units", unitsInput)
unitsInput = Input("units", type="number", placeholder="12")

formItemThreshold = FormControl("Reorder Threshold", thresholdInput)
thresholdInput = Input("threshold", type="number", placeholder="10")

formItemSearch = FormControl("Search Items", searchInput)
searchInput = Input("search", placeholder="Search items...")

reorderButtons = Buttons([
    submitBtn,
    cancelBtn
])

submitBtn = Button("Reorder Selected", null, "primary")
cancelBtn = Button("Cancel", null, "tertiary")

searchInput = Input("search", placeholder="Search items...")
```
