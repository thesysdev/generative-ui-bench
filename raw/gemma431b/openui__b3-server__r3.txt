root = Card([header, regionFilter, healthOverview, metricsSection, incidentsSection, latencySection, alertConfig, footer])

header = CardHeader("Service Health Dashboard", "On-Call Engineer View")

regionFilter = Card([filterText, regionSelect], "sunk", "row", "m", "center", "between")
filterText = TextContent("Filter by Region:", "default")
regionSelect = Select("region_filter", [usEast, usWest, euWest, apSouth], "All Regions", null, "us-east-1", "medium")
usEast = SelectItem("us-east-1", "US East (N. Virginia)")
usWest = SelectItem("us-west-2", "US West (Oregon)")
euWest = SelectItem("eu-west-1", "EU West (Ireland)")
apSouth = SelectItem("ap-south-1", "AP South (Mumbai)")

healthOverview = Card([uptimeStat, errorStat], "card", "row", "m", "stretch", "between")
uptimeStat = TextCallout("success", "Quarterly Uptime", "99.98% - Stable")
errorStat = TextCallout("danger", "Current Error Rate", "0.42% - Spiking")

metricsSection = Card([metricsHeading, resourceChart], "sunk", "column", "s")
metricsHeading = Heading("Resource Usage (24h)", "h3")
resourceChart = LineChart(["-24h", "-18h", "-12h", "-6h", "Now"], [cpuSeries, memSeries], "natural", "Time", "Percentage")
cpuSeries = Series("CPU Usage", [42, 45, 88, 50, 48])
memSeries = Series("Memory Usage", [60, 62, 65, 68, 70])

incidentsSection = Card([incHeading, incidentTable, incidentActions], "card", "column", "m")
incHeading = Heading("Open Incidents", "h3")
incidentTable = Table([incId, incSev, incStatus])
incId = Col("Incident", ["INC-402", "INC-405", "INC-409"], "string")
incSev = Col("Severity", ["Critical", "High", "Medium"], "string")
incStatus = Col("Status", ["Investigating", "Identified", "Monitoring"], "string")
incidentActions = Buttons([ackBtn])
ackBtn = Button("Acknowledge Incident", null, "primary", "normal", "small")

latencySection = Card([latHeading, slowEndpoints], "sunk", "column", "s")
latHeading = Heading("Slowest Endpoints", "h3")
slowEndpoints = Table([endpointName, latencyVal])
endpointName = Col("Endpoint", ["/api/v1/search", "/api/v1/checkout", "/api/v1/user/profile", "/api/v1/auth/verify", "/api/v1/report"], "string")
latencyVal = Col("P99 Latency", ["1.2s", "850ms", "620ms", "410ms", "380ms"], "string")

alertConfig = Card([alertHeading, thresholdForm], "card", "column", "m")
alertHeading = Heading("Alert Thresholds", "h3")
thresholdForm = Form("alert-settings", formBtns, [cpuField, errField])
cpuField = FormControl("CPU Threshold (%)", cpuInput, "Trigger alert if CPU exceeds this value")
cpuInput = Input("cpu_threshold", "80", "number")
errField = FormControl("Error Rate Threshold (%)", errInput, "Trigger alert if error rate exceeds this value")
errInput = Input("err_threshold", "1.0", "number")
formBtns = Buttons([saveBtn])
saveBtn = Button("Save Thresholds", null, "primary")

footer = FollowUpBlock([checkLogs, viewRunbook, contactLead])
checkLogs = FollowUpItem("View detailed system logs")
viewRunbook = FollowUpItem("Open incident runbook")
contactLead = FollowUpItem("Escalate to Engineering Lead")
