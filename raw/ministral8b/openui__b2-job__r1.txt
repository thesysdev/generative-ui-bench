root = Card([roleHeader, roleDescription, qualifications, compensation, applicationForm, shareButton])

roleHeader = CardHeader("Senior UI Designer", "Product Design Team")
roleDescription = TextContent("We're looking for a Senior UI Designer to lead our product design system and elevate our user experience. You'll collaborate with cross-functional teams to create intuitive, beautiful interfaces that drive engagement and business growth.")

qualifications = Card([qualTitle, qualList], "sunk")
qualTitle = Heading("Qualifications", "h3")
qualList = ListBlock([
    ListItem("5+ years of professional design experience", "Specialization in UI/UX design with a strong portfolio"),
    ListItem("Proficient in Figma/Adobe XD", "Experience with design systems and component libraries"),
    ListItem("Collaborative mindset", "Ability to work with developers, product managers, and stakeholders"),
    ListItem("Passion for accessibility", "Commitment to inclusive design principles")
])

compensation = Card([compTitle, compRange, benefits], "sunk")
compTitle = Heading("Compensation", "h3")
compRange = TextContent("$110,000 - $140,000/year", "large-heavy")
benefits = TextContent("Full benefits package including health insurance, 401k matching, and unlimited PTO")

applicationForm = Form("design-application", formButtons, [
    nameField, emailField, resumeField
])
nameField = FormControl("Full Name", nameInput)
nameInput = Input("name", "John Doe", "text")

emailInput = Input("email", "john.doe@company.com", "email")
resumeField = FormControl("Resume Link", resumeInput, "Please provide a link to your resume")
resumeInput = Input("resume", "https://example.com/resume.pdf", "url")

formButtons = Buttons([submitBtn, cancelBtn])
submitBtn = Button("Apply Now", null, "primary")
cancelBtn = Button("Cancel", null, "tertiary")

shareButton = Button("Share Job", null, "secondary")
