```openui-lang
root = Card([newsletterIntro, subscribeForm])

newsletterIntro = CardHeader(
    "Get weekly product updates",
    "Stay ahead with our curated newsletter on the latest features and insights"
)

subscribeForm = Form(
    "newsletter-signup",
    formButtons,
    [emailField]
)

emailField = FormControl(
    "Your email address",
    emailInput,
    "We’ll never share your email with anyone else"
)

emailInput = Input(
    "email",
    "Enter your email",
    "email"
)

formButtons = Buttons([submitButton])

submitButton = Button(
    "Subscribe",
    null,
    "primary"
)
```
