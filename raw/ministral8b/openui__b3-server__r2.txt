root = Card([header, regionFilter, uptimeCard, metricsChart, errorRate, incidentsList, slowEndpoints, alertThreshold, acknowledgeButton])

)

)

    "sunk"
)

    ["Up", "Down"],
    [9998, 2],
    "pie",
    "circular"
)

    [cpuMemTitle, cpuChart, memChart],
    "sunk"
)
cpuMemTitle = Heading("Resource Usage (Last 24h)", "h3")

    ["00:00", "06:00", "12:00", "18:00"],

    "Time (UTC)",
    "CPU Usage (%)"
)

    ["00:00", "06:00", "12:00", "18:00"],
    [memSeries],
    "natural",
    "Time (UTC)",
    "Memory Usage (%)"
)

    [errorTitle, errorValue, errorAlert],
    "sunk"
)

    "sunk"
)

    [slowTitle, slowTable],

)

    [thresholdTitle, thresholdSlider],
    "sunk"
)

    "error_threshold",
    "continuous",
    0,
    5,
    0.5,
    1.5,
    "Error Rate (%)",
    null,
