root = Card([roleHeader, roleDescription, qualifications, salaryInfo, applicationForm, shareButton])

    ListItem("Collaborative mindset", "Ability to work across teams and translate complex requirements"),
    ListItem("Design systems expertise", "Experience building reusable components and design systems"),

])
