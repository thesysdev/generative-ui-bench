RadioItem(label: string, description: string, value: string) = RadioItem(label, description, value)

Buttons(buttons: (Button)[], direction?: "row" | "column") = Buttons(buttons, direction)
Form(name: string, buttons: Buttons, fields: (FormControl)[]) = Form(name, buttons, fields)
FormControl(label: string, input: Input | TextArea | Select | DatePicker | Slider | CheckBoxGroup | RadioGroup, hint?: string) = FormControl(label, input, hint)
Input(name: string, placeholder?: string, type?: "text" | "email" | "password" | "number" | "url", rules?: (string | number)[], value?: string) = Input(name, placeholder, type, rules, value)
TextArea(name: string, placeholder?: string, rows?: number, rules?: (string | number)[], value?: string) = TextArea(name, placeholder, rows, rules, value)
Select(name: string, items: (SelectItem)[], placeholder?: string, rules?: (string | number)[], value?: string, size?: "small" | "medium" | "large") = Select(name, items, placeholder, rules, value, size)

Slider(name: string, variant: "continuous" | "discrete", min: number, max: number, step?: number, defaultValue?: (string | number)[], label?: string, rules?: (string | number)[], value?: (string | number)) = Slider(name, variant, min, max, step, defaultValue, label, rules, value)

CheckBoxItem(label: string, description: string, name: string, defaultChecked?: boolean) = CheckBoxItem(label, description, name, defaultChecked)
RadioGroup(name: string, items: (RadioItem)[], defaultValue?: string, rules?: (string | number)[], value?: string) = RadioGroup(name, items, defaultValue, rules, value)
