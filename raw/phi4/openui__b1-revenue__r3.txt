root = Card([header, revenueToday, comparison, actions])
header = CardHeader("Morning Revenue Report")

revenueToday = Card([revenueHeading, revenueValue])
revenueHeading = Heading("Revenue Today", "h3")
revenueValue = TextContent("$15,200")

comparison = Card([comparisonHeading, comparisonChart])
comparisonHeading = Heading("Compared to Yesterday", "h3")
comparisonChart = LineChart(["8AM", "9AM", "10AM"], [yesterdaySeries, todaySeries], "linear", "Time", "USD")
yesterdaySeries = Series("Yesterday", [5000, 8000, 12000])
todaySeries = Series("Today", [4000, 7000, 14000])

actions = Buttons([refreshBtn])
refreshBtn = Button("Refresh Data")
