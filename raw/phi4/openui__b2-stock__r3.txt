root = Card([inventoryHeader, lowInventoryList, reorderSection, searchField, reorderButtons])

inventoryHeader = CardHeader("Low Inventory Warning", "Items below reorder threshold")

lowInventoryList = Table([itemName, remainingUnits, reorderButton])
itemName = Col("Item", ["Milk", "Bread", "Eggs"])
remainingUnits = Col("Remaining Units", ["5", "3", "2"])
reorderButton = Col("Reorder", [reorderMilk, reorderBread, reorderEggs], "action")
reorderMilk = Button("Reorder Milk")
reorderBread = Button("Reorder Bread")
reorderEggs = Button("Reorder Eggs")

reorderSection = Card([thresholdHeader, reorderThreshold])
thresholdHeader = Heading("Reorder Threshold", "h4")
reorderThreshold = TextContent("Items are low when inventory is below 10 units.")

searchField = Input("search", "Search items...", "text")

reorderButtons = Buttons([searchBtn, resetBtn])
searchBtn = Button("Search")
resetBtn = Button("Reset", null, "tertiary")
