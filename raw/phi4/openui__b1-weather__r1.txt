```openui
root = Card([currentWeather, threeDayOutlook, sunriseSunset])

currentWeather = Card([weatherHeader, temperature, conditions])
weatherHeader = CardHeader("Current Weather in Bengaluru")
temperature = Heading("25°C", "h3")
conditions = TextContent("Sunny", "large")

threeDayOutlook = Card([outlookHeader, day1, day2, day3])
outlookHeader = CardHeader("3-Day Outlook")
day1 = Card([Heading("Day 1", "h4"), Temperature("26°C"), Condition("Cloudy", "small")]
day2 = Card([Heading("Day 2", "h4"), Temperature("27°C"), Condition("Partly Cloudy", "small")]
day3 = Card([Heading("Day 3", "h4"), Temperature("28°C"), Condition("Rain", "small")]

sunriseSunset = Card([sunriseSunsetHeader, times])
sunriseSunsetHeader = CardHeader("Sunrise and Sunset")
times = Table([sunriseCol, sunsetCol])
sunriseCol = Col("Time", ["06:15 AM"])
sunsetCol = Col("Time", ["06:45 PM"])

Temperature = (temp) => TextContent(temp, "large")
Condition = (condition, size) => TextContent(condition, size)
```
