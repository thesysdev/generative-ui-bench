root = Card([question, form])

question = CardHeader("Lunch Poll", "Where should we go for lunch on Friday?")

mexican = RadioItem("Mexican", "Try tacos and burritos", "mexican")
sushi = RadioItem("Sushi", "Fresh sushi and sashimi", "sushi")

formButtons = Buttons([submitBtn, cancelBtn])
submitBtn = Button("Submit vote", null, "primary")
cancelBtn = Button("Cancel", null, "tertiary")
