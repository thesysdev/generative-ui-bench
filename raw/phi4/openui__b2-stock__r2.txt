root = Card([header, inventoryList, reorderSection, searchField])

reorderWidgets = Button("Reorder Widgets", null, "primary")
reorderGadgets = Button("Reorder Gadgets", null, "primary")
reorderThingamajigs = Button("Reorder Thingamajigs", null, "primary")

reorderSection = Card([thresholdHeader, reorderInstructions])
thresholdHeader = Heading("Reorder Threshold", "h4")
reorderInstructions = TextContent("Items with less than 15 units trigger this warning. Use the buttons above to reorder.")

searchField = Form("low-inventory-search", searchButtons, [searchFormControl])
searchFormControl = FormControl("Search Items", searchInput)
searchInput = Input("search", null, "text")
