root = Card([
  visitorsChart,
  conversionRate,
  trafficSources,
  runningCampaigns,
  bestCampaignCallout,
  dateRangeControl,
  subscriberGrowth,
  exportButton
])

visitorsChart = LineChart(["Day 1", "Day 2", ..., "Day 30"], [visitorsSeries], "linear", "Day", "Visitors")
visitorsSeries = Series("Visitors", [1200, 1300, ..., 1500])

conversionRate = Card([conversionHeading, conversionValue])
conversionHeading = Heading("Conversion Rate", "h3")
conversionValue = Tag("3.5%", "success")

trafficSources = Card([sourcesHeading, sourcesTable], "sunk")
sourcesHeading = Heading("Top 5 Traffic Sources", "h3")
sourcesTable = Table([sourceName, sourceShare])
sourceName = Col("Source", ["Organic", "Paid", "Direct", "Email", "Social"])
sourceShare = Col("Share", ["40%", "25%", "15%", "10%", "10%"])

runningCampaigns = Card([campaignsHeading, campaignsTable], "sunk")
campaignsHeading = Heading("Running Campaigns", "h3")
campaignsTable = Table([campaignName, campaignSpend])
campaignName = Col("Campaign", ["Spring Sale", "Winter Campaign", "Summer Special"])
campaignSpend = Col("Spend", ["$5,000", "$4,200", "$3,800"])

bestCampaignCallout = Card([calloutTitle, calloutDescription], "sunk")
calloutTitle = Heading("Best Performing Campaign", "h3")
calloutDescription = Callout("success", "Spring Sale", "Highest ROI this week!")

dateRangeControl = Form("date-range", [applyDateRangeButton], [dateRangeField])
dateRangeField = FormControl("Date Range", dateRangeInput)
dateRangeInput = Input("date_range", "", "range")
applyDateRangeButton = Button("Apply", null, "primary")

subscriberGrowth = Card([growthHeading, growthValue])
growthHeading = Heading("Email Subscriber Growth", "h3")
growthValue = Tag("15%", "success")

exportButton = Button("Export Report", null, "secondary")
