root = Card([

    activeProjectsSection,
    overdueMilestonesSection,
    budgetSection,
    utilizationSection,
    risksSection,
    upcomingDeadlinesSection,
    deliveryPaceSection,
    blockedWorkSection,
    healthSummary,
    quarterChangeButton,
    requestPeopleButton,
    exportReviewButton,
    archiveProjectButton

portfolioHeader = CardHeader("Project Portfolio Review", "Overview of current projects and metrics")

activeProjectsSection = Card([

    activeProjectsTable
], "sunk")

deadlineCol = Col("Deadline", ["2023-11-15", "2023-12-01", "2023-11-30"])
healthCol = Col("Health", ["Good", "Fair", "Critical"])

overdueMilestonesSection = Card([
    overdueMilestonesHeading,
    overdueMilestonesList
], "sunk")

    ListItem("Milestone 1", "Project A", null, "Overdue by 3 days"),
    ListItem("Milestone 2", "Project B", null, "Overdue by 1 week")
], "number")

budgetSection = Card([
    budgetHeading,
    budgetTable
], "sunk")
budgetHeading = Heading("Budget", "h3")
budgetTable = Table([projectBudgetCol, spentCol, allocationCol])

spentCol = Col("Spent", ["$50,000", "$30,000", "$10,000"])
allocationCol = Col("Allocation", ["$100,000", "$80,000", "$50,000"])

    utilizationHeading,
    utilizationChart
], "sunk")
utilizationHeading = Heading("Team Utilization", "h3")

    risksHeading,
    risksTable
], "sunk")

risksTable = Table([riskCol, severityCol])
riskCol = Col("Risk", ["Risk A", "Risk B", "Risk C"])
severityCol = Col("Severity", ["High", "Medium", "Low"])

upcomingDeadlinesSection = Card([
    upcomingDeadlinesHeading,
    upcomingDeadlinesChart
], "sunk")
upcomingDeadlinesHeading = Heading("Upcoming Deadlines", "h3")

    deliveryPaceHeading,
    deliveryPaceChart
], "sunk")

    blockedWorkHeading,
    blockedWorkList
], "sunk")

    ListItem("Blocked Item 1", "Project A", null, "Waiting for design"),
